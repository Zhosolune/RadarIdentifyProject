# 并行任务管道方案

## 1. 方案概述和设计原则

### 1.1 核心设计理念

本方案基于"应用层迭代 + 单元处理管道"的核心思想，在保持增强式管道架构简洁性的前提下，通过应用层的智能调度实现高效并行处理。

### 1.2 设计原则

- **架构稳定性**：不修改PipelineRunner和BaseProcessor核心架构
- **职责分离**：管道负责单元处理，应用层负责批量调度
- **线程安全**：确保并发执行时的数据一致性
- **资源管理**：合理控制线程池和内存使用
- **向后兼容**：保持与现有代码的完全兼容

### 1.3 技术架构

```
┌─ Application Layer ─────────────────────────────────────┐
│ ParallelTaskManager (并行任务管理器)                      │
│ ├─ BatchProcessor (批量处理器)                           │
│ ├─ AsyncTaskScheduler (异步任务调度器)                   │
│ └─ ResourceManager (资源管理器)                          │
└─────────────────────────────────────────────────────────┘
                           ↓ 调用
┌─ Domain Layer ──────────────────────────────────────────┐
│ 现有处理器 (ClusterPlottingProcessor等)                  │
│ 现有服务 (ClusterPlottingService等)                     │
└─────────────────────────────────────────────────────────┘
                           ↓ 使用
┌─ Infrastructure Layer ──────────────────────────────────┐
│ PipelineRunner (保持不变)                               │
│ ThreadPoolExecutor (线程池)                            │
└─────────────────────────────────────────────────────────┘
```

## 2. 场景一：一对多数据流转处理

### 2.1 问题描述

CF/PW聚类处理器输出`List[ClusterResult]`，需要对每个ClusterResult执行完整的处理流程：
```
List[ClusterResult] → 图像生成 → 图像识别 → 特征提取 → 结果保存
```

### 2.2 解决方案：批量处理器

#### 2.2.1 核心实现

```python
"""批量处理器实现"""

import threading
import time
from concurrent.futures import ThreadPoolExecutor, as_completed
from typing import List, Dict, Optional, Callable, Any
from dataclasses import dataclass
from radar_system.domain.recognition.entities.cluster_result import ClusterResult
from radar_system.domain.recognition.services.pipeline_runner import PipelineRunner
from radar_system.infrastructure.config.pipeline_loader import PipelineConfigLoader
from radar_system.infrastructure.common.logging import system_logger

@dataclass
class BatchProcessingResult:
    """批量处理结果"""
    successful_results: List[Dict]
    failed_items: List[Dict]
    total_count: int
    success_rate: float
    processing_time: float

class BatchProcessor:
    """批量处理器
    
    负责将List[ClusterResult]分批并行处理，每个ClusterResult
    通过独立的管道实例进行完整的处理流程。
    """
    
    def __init__(self, 
                 pipeline_config_name: str = "cluster_processing",
                 max_workers: int = 4,
                 batch_size: int = 10):
        """初始化批量处理器
        
        Args:
            pipeline_config_name: 管道配置名称
            max_workers: 最大工作线程数
            batch_size: 批处理大小
        """
        self.pipeline_config_name = pipeline_config_name
        self.max_workers = max_workers
        self.batch_size = batch_size
        
        # 加载管道配置
        self.config_loader = PipelineConfigLoader()
        self.pipeline_config = self.config_loader.get_pipeline_config(pipeline_config_name)
        
        # 线程安全的统计信息
        self._lock = threading.Lock()
        self._stats = {
            "total_processed": 0,
            "total_failed": 0,
            "average_processing_time": 0.0
        }
        
        system_logger.info(f"批量处理器初始化完成: max_workers={max_workers}, batch_size={batch_size}")
    
    def process_clusters(self, 
                        clusters: List[ClusterResult],
                        progress_callback: Optional[Callable[[int, int], None]] = None) -> BatchProcessingResult:
        """批量处理聚类结果
        
        Args:
            clusters: 聚类结果列表
            progress_callback: 进度回调函数 (current, total)
            
        Returns:
            BatchProcessingResult: 批量处理结果
        """
        start_time = time.time()
        
        if not clusters:
            return BatchProcessingResult([], [], 0, 0.0, 0.0)
        
        system_logger.info(f"开始批量处理 {len(clusters)} 个聚类结果")
        
        successful_results = []
        failed_items = []
        
        # 创建批次
        batches = self._create_batches(clusters, self.batch_size)
        
        with ThreadPoolExecutor(max_workers=self.max_workers) as executor:
            # 提交所有批次任务
            future_to_batch = {
                executor.submit(self._process_batch, batch, batch_idx): batch_idx
                for batch_idx, batch in enumerate(batches)
            }
            
            processed_count = 0
            
            # 收集结果
            for future in as_completed(future_to_batch):
                batch_idx = future_to_batch[future]
                
                try:
                    batch_results, batch_failures = future.result()
                    successful_results.extend(batch_results)
                    failed_items.extend(batch_failures)
                    
                    processed_count += len(batches[batch_idx])
                    
                    # 调用进度回调
                    if progress_callback:
                        progress_callback(processed_count, len(clusters))
                    
                    system_logger.debug(f"批次 {batch_idx} 处理完成")
                    
                except Exception as e:
                    # 整个批次失败
                    batch = batches[batch_idx]
                    for cluster in batch:
                        failed_items.append({
                            "cluster_index": cluster.cluster_index,
                            "error": f"批次处理失败: {str(e)}"
                        })
                    
                    system_logger.error(f"批次 {batch_idx} 处理失败: {str(e)}")
        
        # 计算结果统计
        processing_time = time.time() - start_time
        success_rate = len(successful_results) / len(clusters) if clusters else 0.0
        
        # 更新统计信息
        self._update_stats(len(clusters), len(failed_items), processing_time)
        
        result = BatchProcessingResult(
            successful_results=successful_results,
            failed_items=failed_items,
            total_count=len(clusters),
            success_rate=success_rate,
            processing_time=processing_time
        )
        
        system_logger.info(
            f"批量处理完成: 成功 {len(successful_results)}/{len(clusters)} 个, "
            f"成功率 {success_rate:.2%}, 耗时 {processing_time:.2f}s"
        )
        
        return result
    
    def _process_batch(self, batch: List[ClusterResult], batch_idx: int) -> tuple[List[Dict], List[Dict]]:
        """处理单个批次
        
        Args:
            batch: 批次中的聚类结果列表
            batch_idx: 批次索引
            
        Returns:
            tuple: (成功结果列表, 失败项列表)
        """
        successful_results = []
        failed_items = []
        
        for cluster in batch:
            try:
                # 为每个聚类创建独立的管道实例
                runner = PipelineRunner(self.pipeline_config)
                result = runner.run({"cluster_result": cluster})
                
                successful_results.append({
                    "cluster_index": cluster.cluster_index,
                    "cluster_dim": cluster.dim_name,
                    "processing_result": result
                })
                
            except Exception as e:
                failed_items.append({
                    "cluster_index": cluster.cluster_index,
                    "error": str(e)
                })
                
                system_logger.warning(f"聚类 {cluster.cluster_index} 处理失败: {str(e)}")
        
        return successful_results, failed_items
    
    def _create_batches(self, items: List[Any], batch_size: int) -> List[List[Any]]:
        """创建批次"""
        return [items[i:i + batch_size] for i in range(0, len(items), batch_size)]
    
    def _update_stats(self, total: int, failed: int, processing_time: float) -> None:
        """更新统计信息"""
        with self._lock:
            self._stats["total_processed"] += total
            self._stats["total_failed"] += failed
            
            # 计算平均处理时间
            current_avg = self._stats["average_processing_time"]
            new_avg = (current_avg + processing_time) / 2
            self._stats["average_processing_time"] = new_avg
    
    def get_statistics(self) -> Dict:
        """获取处理统计信息"""
        with self._lock:
            return self._stats.copy()
```

## 3. 场景二：高吞吐量任务的并行处理

### 3.1 问题描述

当处理器执行密集型任务（如数据保存为Excel文件）时，需要让下一级处理器并行运行而不互相影响：
```
处理器A (耗时任务) ║ 处理器B (下一步处理)
     ↓             ║      ↓
保存Excel文件      ║  特征提取/分析
(5-10秒)          ║  (并行执行)
```

### 3.2 解决方案：异步任务调度器

#### 3.2.1 核心实现

```python
"""异步任务调度器实现"""

import asyncio
import threading
import queue
import time
from concurrent.futures import ThreadPoolExecutor, Future
from typing import Dict, List, Optional, Callable, Any
from dataclasses import dataclass, field
from enum import Enum

class TaskPriority(Enum):
    """任务优先级"""
    LOW = 1
    NORMAL = 2
    HIGH = 3
    CRITICAL = 4

@dataclass
class AsyncTask:
    """异步任务定义"""
    task_id: str
    task_type: str
    processor_name: str
    input_data: Dict
    params: Dict
    priority: TaskPriority = TaskPriority.NORMAL
    callback: Optional[Callable] = None
    created_time: float = field(default_factory=time.time)

class TaskResult:
    """任务结果"""
    def __init__(self, task_id: str, success: bool, result: Any = None, error: str = None):
        self.task_id = task_id
        self.success = success
        self.result = result
        self.error = error
        self.completed_time = time.time()

class AsyncTaskScheduler:
    """异步任务调度器

    支持高吞吐量任务的并行处理，允许耗时任务与后续处理并行执行。
    """

    def __init__(self,
                 max_workers: int = 6,
                 max_queue_size: int = 100,
                 enable_priority: bool = True):
        """初始化异步任务调度器

        Args:
            max_workers: 最大工作线程数
            max_queue_size: 最大队列大小
            enable_priority: 是否启用优先级调度
        """
        self.max_workers = max_workers
        self.max_queue_size = max_queue_size
        self.enable_priority = enable_priority

        # 线程池
        self.executor = ThreadPoolExecutor(max_workers=max_workers)

        # 任务队列 (支持优先级)
        if enable_priority:
            self.task_queue = queue.PriorityQueue(maxsize=max_queue_size)
        else:
            self.task_queue = queue.Queue(maxsize=max_queue_size)

        # 结果存储
        self.results: Dict[str, TaskResult] = {}
        self.active_tasks: Dict[str, Future] = {}

        # 线程安全锁
        self._lock = threading.RLock()

        # 调度器状态
        self._running = False
        self._scheduler_thread = None

        # 统计信息
        self._stats = {
            "total_submitted": 0,
            "total_completed": 0,
            "total_failed": 0,
            "average_execution_time": 0.0
        }

        system_logger.info(f"异步任务调度器初始化完成: max_workers={max_workers}")

    def start(self) -> None:
        """启动调度器"""
        if self._running:
            return

        self._running = True
        self._scheduler_thread = threading.Thread(target=self._scheduler_loop, daemon=True)
        self._scheduler_thread.start()

        system_logger.info("异步任务调度器已启动")

    def stop(self) -> None:
        """停止调度器"""
        self._running = False

        if self._scheduler_thread:
            self._scheduler_thread.join(timeout=5.0)

        self.executor.shutdown(wait=True)
        system_logger.info("异步任务调度器已停止")

    def submit_task(self, task: AsyncTask) -> str:
        """提交异步任务

        Args:
            task: 异步任务对象

        Returns:
            str: 任务ID

        Raises:
            queue.Full: 队列已满
        """
        try:
            if self.enable_priority:
                # 优先级队列：数值越小优先级越高
                priority_value = 5 - task.priority.value
                self.task_queue.put((priority_value, task.created_time, task), timeout=1.0)
            else:
                self.task_queue.put(task, timeout=1.0)

            with self._lock:
                self._stats["total_submitted"] += 1

            system_logger.debug(f"任务 {task.task_id} 已提交到队列")
            return task.task_id

        except queue.Full:
            raise RuntimeError(f"任务队列已满，无法提交任务 {task.task_id}")

    def get_result(self, task_id: str, timeout: Optional[float] = None) -> Optional[TaskResult]:
        """获取任务结果

        Args:
            task_id: 任务ID
            timeout: 超时时间（秒）

        Returns:
            TaskResult: 任务结果，如果任务未完成则返回None
        """
        start_time = time.time()

        while True:
            with self._lock:
                if task_id in self.results:
                    return self.results[task_id]

            # 检查超时
            if timeout and (time.time() - start_time) > timeout:
                return None

            time.sleep(0.1)  # 短暂等待

    def _scheduler_loop(self) -> None:
        """调度器主循环"""
        while self._running:
            try:
                # 获取任务
                if self.enable_priority:
                    _, _, task = self.task_queue.get(timeout=1.0)
                else:
                    task = self.task_queue.get(timeout=1.0)

                # 提交到线程池执行
                future = self.executor.submit(self._execute_task, task)

                with self._lock:
                    self.active_tasks[task.task_id] = future

                # 设置完成回调
                future.add_done_callback(lambda f, tid=task.task_id: self._task_completed(tid, f))

            except queue.Empty:
                continue
            except Exception as e:
                system_logger.error(f"调度器循环出错: {str(e)}")

    def _execute_task(self, task: AsyncTask) -> TaskResult:
        """执行单个任务"""
        start_time = time.time()

        try:
            # 获取处理器并执行
            from radar_system.domain.recognition.processors.registry import get_processor

            processor_cls = get_processor(task.processor_name)
            processor = processor_cls()

            # 执行处理
            result = processor.run(task.input_data, task.params)

            execution_time = time.time() - start_time

            # 更新统计信息
            self._update_execution_stats(execution_time, success=True)

            task_result = TaskResult(task.task_id, True, result)

            # 调用回调函数
            if task.callback:
                try:
                    task.callback(task_result)
                except Exception as e:
                    system_logger.warning(f"任务 {task.task_id} 回调执行失败: {str(e)}")

            return task_result

        except Exception as e:
            execution_time = time.time() - start_time
            self._update_execution_stats(execution_time, success=False)

            error_msg = f"任务执行失败: {str(e)}"
            system_logger.error(f"任务 {task.task_id} {error_msg}")

            return TaskResult(task.task_id, False, error=error_msg)

    def _task_completed(self, task_id: str, future: Future) -> None:
        """任务完成回调"""
        try:
            result = future.result()

            with self._lock:
                self.results[task_id] = result
                if task_id in self.active_tasks:
                    del self.active_tasks[task_id]

                if result.success:
                    self._stats["total_completed"] += 1
                else:
                    self._stats["total_failed"] += 1

            system_logger.debug(f"任务 {task_id} 完成: {'成功' if result.success else '失败'}")

        except Exception as e:
            system_logger.error(f"任务 {task_id} 完成回调出错: {str(e)}")

    def _update_execution_stats(self, execution_time: float, success: bool) -> None:
        """更新执行统计"""
        with self._lock:
            current_avg = self._stats["average_execution_time"]
            total_completed = self._stats["total_completed"] + self._stats["total_failed"]

            if total_completed > 0:
                new_avg = (current_avg * total_completed + execution_time) / (total_completed + 1)
                self._stats["average_execution_time"] = new_avg

    def get_statistics(self) -> Dict:
        """获取调度器统计信息"""
        with self._lock:
            active_count = len(self.active_tasks)
            queue_size = self.task_queue.qsize()

            return {
                **self._stats.copy(),
                "active_tasks": active_count,
                "queue_size": queue_size,
                "success_rate": (
                    self._stats["total_completed"] /
                    max(1, self._stats["total_completed"] + self._stats["total_failed"])
                )
            }
```

#### 3.2.2 并行管道运行器

```python
"""支持异步处理的并行管道运行器"""

class ParallelPipelineRunner:
    """并行管道运行器

    支持管道内步骤的异步并行执行。
    """

    def __init__(self, config: Dict, async_scheduler: Optional[AsyncTaskScheduler] = None):
        """初始化并行管道运行器

        Args:
            config: 管道配置
            async_scheduler: 异步任务调度器
        """
        if "pipeline" not in config:
            raise KeyError("配置中缺少 'pipeline' 键")

        self.steps: List[Dict] = config["pipeline"]
        self.cache: Dict[str, Any] = {}
        self.async_scheduler = async_scheduler

        # 分析步骤依赖关系
        self.step_dependencies = self._analyze_dependencies()

        system_logger.info(f"并行管道运行器初始化完成，包含 {len(self.steps)} 个步骤")

    def run(self, initial_data: Optional[Dict] = None) -> Dict:
        """执行管道流程（支持并行）"""
        if initial_data:
            self.cache["input"] = initial_data

        # 如果没有异步调度器，回退到顺序执行
        if not self.async_scheduler:
            return self._run_sequential()

        return self._run_parallel()

    def _run_parallel(self) -> Dict:
        """并行执行模式"""
        completed_steps = set()
        step_futures = {}

        while len(completed_steps) < len(self.steps):
            # 查找可以执行的步骤
            ready_steps = self._get_ready_steps(completed_steps)

            # 提交就绪的步骤
            for step in ready_steps:
                if step["name"] not in step_futures:
                    task = self._create_async_task(step)
                    task_id = self.async_scheduler.submit_task(task)
                    step_futures[step["name"]] = task_id

            # 检查已完成的步骤
            for step_name, task_id in list(step_futures.items()):
                result = self.async_scheduler.get_result(task_id, timeout=0.1)
                if result:
                    if result.success:
                        self.cache[step_name] = result.result
                        completed_steps.add(step_name)
                        system_logger.info(f"步骤 '{step_name}' 完成")
                    else:
                        error_msg = f"步骤 '{step_name}' 失败: {result.error}"
                        system_logger.error(error_msg)
                        raise RuntimeError(error_msg)

                    del step_futures[step_name]

            time.sleep(0.1)  # 短暂等待

        return self.cache

    def _get_ready_steps(self, completed_steps: set) -> List[Dict]:
        """获取可以执行的步骤"""
        ready_steps = []

        for step in self.steps:
            step_name = step["name"]

            # 跳过已完成的步骤
            if step_name in completed_steps:
                continue

            # 检查依赖是否满足
            dependencies = step.get("input_from", [])
            if all(dep in completed_steps or dep == "input" for dep in dependencies):
                ready_steps.append(step)

        return ready_steps

    def _create_async_task(self, step: Dict) -> AsyncTask:
        """创建异步任务"""
        step_name = step["name"]
        processor_name = step["processor"]

        # 准备输入数据
        input_data = self._prepare_input_data(step)

        # 确定任务优先级
        priority = TaskPriority.NORMAL
        if step.get("high_priority", False):
            priority = TaskPriority.HIGH
        elif step.get("low_priority", False):
            priority = TaskPriority.LOW

        return AsyncTask(
            task_id=f"{step_name}_{int(time.time() * 1000)}",
            task_type="pipeline_step",
            processor_name=processor_name,
            input_data=input_data,
            params=step.get("params", {}),
            priority=priority
        )

        """异步导出结果

            results: 处理结果列表
            export_path: 导出路径
            callback: 完成回调函数

        Returns:
            str: 任务ID
        """
        # 创建异步任务
        task = AsyncTask(
            task_id=f"export_{int(time.time() * 1000)}",
            task_type="excel_export",
            processor_name="excel_export_processor",
            input_data={"results": results},
            params={"export_path": export_path},
            priority=TaskPriority.NORMAL,
            callback=callback
        )

        # 提交任务
        task_id = self.async_scheduler.submit_task(task)
        system_logger.info(f"Excel导出任务已提交: {task_id}")

        return task_id

    def process_with_parallel_pipeline(self, data: Dict) -> Dict:
        """使用并行管道处理数据

        支持耗时任务与后续处理并行执行。
        """
        # 执行并行管道
        result = self.pipeline_runner.run(data)
        return result

    def cleanup(self):
        """清理资源"""
        self.async_scheduler.stop()
```

### 5.2 资源监控

```python
"""资源监控实现"""

import psutil
import threading
import time
from typing import Dict, Optional

class ResourceMonitor:
    """资源监控器"""

    def __init__(self, check_interval: float = 5.0):
        self.check_interval = check_interval
        self._monitoring = False
        self._monitor_thread = None
        self._stats = {
            "cpu_percent": 0.0,
            "memory_percent": 0.0,
            "active_threads": 0
        }
        self._lock = threading.Lock()

    def start_monitoring(self):
        """开始监控"""
        if self._monitoring:
            return

        self._monitoring = True
        self._monitor_thread = threading.Thread(target=self._monitor_loop, daemon=True)
        self._monitor_thread.start()

    def stop_monitoring(self):
        """停止监控"""
        self._monitoring = False
        if self._monitor_thread:
            self._monitor_thread.join()

    def _monitor_loop(self):
        """监控循环"""
        while self._monitoring:
            try:
                cpu_percent = psutil.cpu_percent(interval=1)
                memory_percent = psutil.virtual_memory().percent
                active_threads = threading.active_count()

                with self._lock:
                    self._stats.update({
                        "cpu_percent": cpu_percent,
                        "memory_percent": memory_percent,
                        "active_threads": active_threads
                    })

                # 资源告警
                if cpu_percent > 80:
                    system_logger.warning(f"CPU使用率过高: {cpu_percent:.1f}%")
                if memory_percent > 85:
                    system_logger.warning(f"内存使用率过高: {memory_percent:.1f}%")

                time.sleep(self.check_interval)

            except Exception as e:
                system_logger.error(f"资源监控出错: {str(e)}")

    def get_stats(self) -> Dict:
        """获取资源统计"""
        with self._lock:
            return self._stats.copy()

    def is_resource_available(self, cpu_threshold: float = 80.0,
                            memory_threshold: float = 85.0) -> bool:
        """检查资源是否可用"""
        stats = self.get_stats()
        return (stats["cpu_percent"] < cpu_threshold and
                stats["memory_percent"] < memory_threshold)
```

```python
"""性能监控示例"""

class PerformanceProfiler:
    """性能分析器"""

    def __init__(self):
        self.execution_times = {}
        self.error_counts = {}
        self._lock = threading.Lock()

    def profile_execution(self, operation_name: str):
        """执行时间分析装饰器"""
        def decorator(func):
            def wrapper(*args, **kwargs):
                start_time = time.time()
                try:
                    result = func(*args, **kwargs)
                    execution_time = time.time() - start_time
                    self._record_execution(operation_name, execution_time)
                    return result
                except Exception as e:
                    self._record_error(operation_name)
                    raise
            return wrapper
        return decorator

    def _record_execution(self, operation: str, time_taken: float):
        with self._lock:
            if operation not in self.execution_times:
                self.execution_times[operation] = []
            self.execution_times[operation].append(time_taken)

    def _record_error(self, operation: str):
        with self._lock:
            self.error_counts[operation] = self.error_counts.get(operation, 0) + 1

    def get_report(self) -> Dict:
        """生成性能报告"""
        with self._lock:
            report = {}
            for operation, times in self.execution_times.items():
                report[operation] = {
                    "count": len(times),
                    "avg_time": sum(times) / len(times),
                    "min_time": min(times),
                    "max_time": max(times),
                    "error_count": self.error_counts.get(operation, 0)
                }
            return report
```

"""向后兼容的服务接口"""

class EnhancedRecognitionService:
    """增强的识别服务 - 向后兼容"""

    def __init__(self, enable_parallel: bool = True):
        self.enable_parallel = enable_parallel

        if enable_parallel:
            self.batch_processor = BatchProcessor()
            self.async_scheduler = AsyncTaskScheduler()
            self.async_scheduler.start()
        else:
            self.batch_processor = None
            self.async_scheduler = None

    def process_clustering_results(self,
                                 cf_clusters: List[ClusterResult],
                                 pw_clusters: List[ClusterResult]) -> Dict:
        """处理聚类结果 - 兼容接口"""
        if self.enable_parallel and self.batch_processor:
            # 使用并行处理
            return self._process_parallel(cf_clusters + pw_clusters)
        else:
            # 回退到原有的顺序处理
            return self._process_sequential(cf_clusters + pw_clusters)

    def _process_parallel(self, clusters: List[ClusterResult]) -> Dict:
        """并行处理实现"""
        result = self.batch_processor.process_clusters(clusters)
        return {
            "total_clusters": result.total_count,
            "successful_count": len(result.successful_results),
            "success_rate": result.success_rate,
            "results": result.successful_results
        }

    def _process_sequential(self, clusters: List[ClusterResult]) -> Dict:
        """顺序处理实现（原有逻辑）"""
        results = []
        for cluster in clusters:
            # 原有的单个处理逻辑
            runner = PipelineRunner(self.cluster_pipeline_config)
            result = runner.run({"cluster_result": cluster})
            results.append(result)

        return {
            "total_clusters": len(clusters),
            "successful_count": len(results),
            "success_rate": 1.0,
            "results": results
        }
